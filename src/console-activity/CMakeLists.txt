cmake_minimum_required(VERSION 3.16)
project(console_activity VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Activity storage requires the qualified Linux card pin")
endif()
if(NOT TARGET sessiond_linux)
    set(SESSIOND_BUILD_DAEMON ON CACHE BOOL "Qualified Linux storage adapters")
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../console-sessiond"
                     "${CMAKE_CURRENT_BINARY_DIR}/sessiond" EXCLUDE_FROM_ALL)
endif()
add_library(console_activity_sqlite STATIC vendor/sqlite3/sqlite3.c)
target_include_directories(console_activity_sqlite PUBLIC vendor/sqlite3)
target_compile_definitions(console_activity_sqlite PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64
    SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_SHARED_CACHE
    SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_DQS=0 SQLITE_TEMP_STORE=3
    SQLITE_DEFAULT_CACHE_SIZE=-512 SQLITE_MAX_LENGTH=1048576 SQLITE_MAX_SQL_LENGTH=8192)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(console_activity_sqlite PRIVATE -w)
endif()
add_library(console_activity STATIC src/activity.c src/database.c src/storage.c src/page.c)
target_include_directories(console_activity PUBLIC include PRIVATE src)
target_compile_definitions(console_activity PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(console_activity PUBLIC sessiond_linux PRIVATE console_activity_sqlite)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(console_activity PRIVATE -Wall -Wextra -Wpedantic -Wconversion
        -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef -Werror)
endif()
option(ACTIVITY_BUILD_TESTS "Build isolated storage/database regression fixtures" OFF)
if(ACTIVITY_BUILD_TESTS)
    add_executable(activity_tests tests/test_activity.c)
    target_link_libraries(activity_tests PRIVATE console_activity console_activity_sqlite)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(activity_tests PRIVATE -UNDEBUG)
    endif()
endif()
