function(configure_bridge target)
    target_include_directories(${target} PUBLIC . support)
    target_compile_definitions(${target} PUBLIC _GNU_SOURCE _FILE_OFFSET_BITS=64)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow
        -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef -Werror)
endfunction()
add_library(magicx_bridge STATIC ${BRIDGE_SOURCES})
configure_bridge(magicx_bridge)

if(RETROARCH_SOURCE)
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${RETROARCH_SOURCE}" rev-parse HEAD
        OUTPUT_VARIABLE RETROARCH_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE PIN_RESULT)
    if(NOT PIN_RESULT EQUAL 0 OR NOT RETROARCH_HEAD STREQUAL "0792144fe3a7b59908b0afdb2c01722e79040360")
        message(FATAL_ERROR "RETROARCH_SOURCE must use the exact pinned v1.19.1 commit")
    endif()
    add_library(magicx_retroarch_binding OBJECT "${RETROARCH_SOURCE}/magicx/magicx_retroarch.c"
        "${SESSIOND_SOURCE}/src/disc_playlist.c")
    target_include_directories(magicx_retroarch_binding PRIVATE "${RETROARCH_SOURCE}/libretro-common/include"
        "${RETROARCH_SOURCE}/magicx/support" "${SESSIOND_SOURCE}/include")
    target_compile_definitions(magicx_retroarch_binding PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64 HAVE_THREADS HAVE_DYNAMIC HAVE_MAGICX_BRIDGE)
    target_compile_options(magicx_retroarch_binding PRIVATE -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
endif()
