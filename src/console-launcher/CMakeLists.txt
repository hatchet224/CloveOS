cmake_minimum_required(VERSION 3.16)
project(console_launcher VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The launcher uses the board's Linux SDL2 and mounted-card adapters")
endif()

set(SESSIOND_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../console-sessiond" CACHE PATH
    "Matching daemon source directory")
if(NOT TARGET sessiond_linux)
    set(SESSIOND_BUILD_DAEMON ON CACHE BOOL "Provide daemon Linux storage adapters")
    if(NOT SESSIOND_BUILD_DAEMON)
        message(FATAL_ERROR "Launcher history requires SESSIOND_BUILD_DAEMON=ON")
    endif()
    add_subdirectory("${SESSIOND_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/sessiond" EXCLUDE_FROM_ALL)
endif()

add_library(console_launcher_history STATIC src/history.c src/qualification.c src/paging.c src/disc_groups.c src/preference_record.c src/i18n.c)
target_include_directories(console_launcher_history PUBLIC include)
target_compile_definitions(console_launcher_history PUBLIC _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(console_launcher_history PUBLIC sessiond_linux)

find_package(SDL2 2.0.5 CONFIG QUIET)
if(TARGET SDL2::SDL2)
    set(LAUNCHER_SDL_TARGET SDL2::SDL2)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LAUNCHER_SDL REQUIRED IMPORTED_TARGET "sdl2>=2.0.5")
    set(LAUNCHER_SDL_TARGET PkgConfig::LAUNCHER_SDL)
endif()
find_package(SDL2_ttf CONFIG QUIET HINTS "${SDL2_DIR}")
if(TARGET SDL2_ttf::SDL2_ttf)
    set(LAUNCHER_TTF_TARGET SDL2_ttf::SDL2_ttf)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LAUNCHER_TTF REQUIRED IMPORTED_TARGET "SDL2_ttf>=2.0.14")
    set(LAUNCHER_TTF_TARGET PkgConfig::LAUNCHER_TTF)
endif()
add_library(console_launcher STATIC src/app.c src/favorites.c src/preferences.c src/display.c src/input.c src/text.c src/theme.c src/theme_file.c src/theme_sound.c src/assets.c src/view.c src/tools.c vendor/lodepng.c vendor/minimp3.c)
add_library(console_image_decoder STATIC ../console-supervisor/vendor/image_decoder.c)
target_compile_options(console_image_decoder PRIVATE -w)
target_include_directories(console_image_decoder PUBLIC ../console-supervisor/vendor)
target_include_directories(console_launcher PRIVATE vendor)
target_compile_definitions(console_launcher PRIVATE LODEPNG_NO_COMPILE_ENCODER LODEPNG_NO_COMPILE_DISK)
set_source_files_properties(vendor/lodepng.c vendor/minimp3.c PROPERTIES COMPILE_OPTIONS "-w")
target_link_libraries(console_launcher PUBLIC console_launcher_history console_image_decoder ${LAUNCHER_SDL_TARGET} ${LAUNCHER_TTF_TARGET})
foreach(target console_launcher_history console_launcher)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow
            -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef -Werror)
    endif()
endforeach()

# Source fixtures only until explicitly built in a suitable Linux/SDL environment.
option(LAUNCHER_BUILD_TESTS "Build launcher regression fixtures" OFF)
if(LAUNCHER_BUILD_TESTS)
    enable_testing()
    add_executable(launcher_history_tests tests/test_history.c)
    target_link_libraries(launcher_history_tests PRIVATE console_launcher_history)
    add_executable(launcher_input_tests tests/test_input.c)
    target_link_libraries(launcher_input_tests PRIVATE console_launcher)
    add_executable(launcher_favorites_tests tests/test_favorites.c)
    target_link_libraries(launcher_favorites_tests PRIVATE console_launcher)
    add_test(NAME launcher_favorites COMMAND launcher_favorites_tests "${CMAKE_CURRENT_BINARY_DIR}/favorites-fixture")
    # Requires an isolated prepared card root; the target VM runner supplies FAT/tmpfs.
    add_executable(launcher_preferences_tests tests/test_preferences.c src/preferences.c)
    target_compile_definitions(launcher_preferences_tests PRIVATE CL_PREFERENCES_TESTING)
    target_link_libraries(launcher_preferences_tests PRIVATE console_launcher)
    foreach(test_target launcher_history_tests launcher_input_tests launcher_favorites_tests launcher_preferences_tests)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${test_target} PRIVATE -UNDEBUG)
        endif()
    endforeach()
    add_test(NAME launcher_history COMMAND launcher_history_tests)
    add_test(NAME launcher_input COMMAND launcher_input_tests)
    set_tests_properties(launcher_input PROPERTIES SKIP_RETURN_CODE 77)
endif()

option(CLOVE_INSTALL_THEME_ASSETS "Install separately acquired and licensed theme assets" OFF)
if(CLOVE_INSTALL_THEME_ASSETS)
    install(DIRECTORY themes/Clove themes/Silky themes/TMNT DESTINATION share/console/themes)
endif()
