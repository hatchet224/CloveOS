cmake_minimum_required(VERSION 3.16)
project(clove_file_explorer VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "File Explorer requires Linux openat/renameat2 and the qualified mounted-card adapters")
endif()
add_library(clove_file_operations STATIC fs.c)
target_include_directories(clove_file_operations PUBLIC .)
target_compile_definitions(clove_file_operations PUBLIC _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_compile_options(clove_file_operations PRIVATE -Wall -Wextra -Werror)
option(FM_BUILD_APP "Build the managed SDL controller application" ON)
option(FM_BUILD_TESTS "Build isolated filesystem regression tests" OFF)
if(FM_BUILD_APP)
    set(SESSIOND_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../console-sessiond" CACHE PATH "Matching CloveOS session daemon source")
    if(NOT TARGET sessiond_linux)
        set(SESSIOND_BUILD_DAEMON ON CACHE BOOL "Build matching Linux storage adapters")
        add_subdirectory("${SESSIOND_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/sessiond" EXCLUDE_FROM_ALL)
    endif()
    find_package(SDL2 CONFIG QUIET HINTS "${SDL2_DIR}")
    if(TARGET SDL2::SDL2)
        set(EXPLORER_SDL_TARGET SDL2::SDL2)
    elseif(TARGET PkgConfig::LAUNCHER_SDL)
        set(EXPLORER_SDL_TARGET PkgConfig::LAUNCHER_SDL)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(EXPLORER_SDL REQUIRED IMPORTED_TARGET "sdl2>=2.0.18")
        set(EXPLORER_SDL_TARGET PkgConfig::EXPLORER_SDL)
    endif()
    add_executable(console-file-explorer main.c ../console-launcher/src/display.c
        ../console-launcher/src/input.c ../console-launcher/src/text.c)
    target_include_directories(console-file-explorer PRIVATE ../console-launcher/include)
    target_compile_definitions(console-file-explorer PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
    target_compile_options(console-file-explorer PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(console-file-explorer PRIVATE clove_file_operations sessiond_linux ${EXPLORER_SDL_TARGET})
    install(TARGETS console-file-explorer RUNTIME DESTINATION bin)
endif()
if(FM_BUILD_TESTS)
    enable_testing()
    add_executable(file_explorer_fs_tests test_fs.c)
    target_link_libraries(file_explorer_fs_tests PRIVATE clove_file_operations)
    target_compile_options(file_explorer_fs_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME file_explorer_files COMMAND file_explorer_fs_tests)
endif()
