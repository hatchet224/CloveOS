cmake_minimum_required(VERSION 3.16)
project(console_appliance VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    # Custom cross toolchains may leave CMake's release flags empty. Apply
    # optimization explicitly to production release configurations and children.
    add_compile_options($<$<CONFIG:MinSizeRel>:-Os> $<$<CONFIG:Release>:-O2>)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Tina/Moss Linux sources; use the matching qualified target environment")
endif()
option(BUILD_TESTING "Build source regression fixtures" OFF)
set(SESSIOND_BUILD_DAEMON ON CACHE BOOL "Provide daemon Linux adapters")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../console-launcher"
                 "${CMAKE_CURRENT_BINARY_DIR}/launcher" EXCLUDE_FROM_ALL)

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../console-activity"
                 "${CMAKE_CURRENT_BINARY_DIR}/activity" EXCLUDE_FROM_ALL)
add_library(console_online_codec STATIC ../console-launcher/vendor/lodepng.c)
target_link_libraries(console_online_codec PUBLIC console_image_decoder)
target_include_directories(console_online_codec PUBLIC vendor ../console-launcher/vendor)
target_compile_definitions(console_online_codec PRIVATE LODEPNG_NO_COMPILE_DISK)
target_compile_options(console_online_codec PRIVATE -w)
add_library(console_control STATIC src/wire.c src/process.c src/mount.c src/cpu_policy.c src/clock.c src/timezone.c
            src/package_registry.c src/package_manager.c src/network.c src/transfer.c src/online.c src/api_document.c src/online_art.c src/zip.c src/elf_dependency.c src/core_choice.c src/maintenance.c src/glo_scripts.c src/battery_monitor.c)
target_link_libraries(console_control PUBLIC console_online_codec)
set(CONSOLE_ONLINE_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../../work/build-download/prefix" CACHE PATH
    "AArch64 static libcurl/Mbed TLS installation built for the same libc")
target_include_directories(console_control PRIVATE "${CONSOLE_ONLINE_PREFIX}/include")
foreach(dependency curl mbedtls mbedx509 mbedcrypto)
    if(NOT EXISTS "${CONSOLE_ONLINE_PREFIX}/lib/lib${dependency}.a")
        message(FATAL_ERROR "Missing target online dependency lib${dependency}.a in CONSOLE_ONLINE_PREFIX")
    endif()
    target_link_libraries(console_control PUBLIC "${CONSOLE_ONLINE_PREFIX}/lib/lib${dependency}.a")
endforeach()
target_include_directories(console_control PUBLIC include)
target_link_libraries(console_control PUBLIC console_launcher_history console_activity)
add_library(console_catalog_refresh OBJECT src/catalog.c)
target_compile_definitions(console_catalog_refresh PRIVATE SC_CATALOG_LIBRARY _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_include_directories(console_catalog_refresh PRIVATE include ../console-sessiond/include ../console-launcher/include ../console-activity/include)
target_compile_options(console_catalog_refresh PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
target_sources(console_control PRIVATE $<TARGET_OBJECTS:console_catalog_refresh>)

target_sources(console-sessiond PRIVATE src/managed.c)
target_include_directories(console-sessiond PRIVATE include)
target_compile_definitions(console-sessiond PRIVATE HAVE_CONSOLE_SUPERVISOR)
target_link_libraries(console-sessiond PRIVATE console_control)

add_executable(console-supervisor src/main.c src/config.c src/disk.c src/catalog_index.c src/diagnostics.c src/tools_backend.c src/file_service.c src/lan_services.c)
target_link_libraries(console-supervisor PRIVATE console_control)
add_dependencies(console-supervisor console-sessiond)
add_executable(console-launcher src/ui_host.c)
target_link_libraries(console-launcher PRIVATE console_control console_launcher)
# Distinct supervised entrypoint; requires the same authenticated FD3 handshake.
add_executable(GameSwitcher src/ui_host.c)
target_compile_definitions(GameSwitcher PRIVATE CL_START_GAMESWITCHER=1)
target_link_libraries(GameSwitcher PRIVATE console_control console_launcher)
add_executable(console-catalog src/catalog.c src/config.c src/catalog_index.c)
target_link_libraries(console-catalog PRIVATE console_control)
add_executable(five_apps_controller_tests ../console-launcher/tests/test_five_apps.c)
target_link_libraries(five_apps_controller_tests PRIVATE console_launcher)
target_compile_options(five_apps_controller_tests PRIVATE -UNDEBUG)
add_executable(five_apps_backend_tests tests/test_five_apps_backend.c src/config.c src/catalog_index.c src/diagnostics.c src/tools_backend.c)
target_link_libraries(five_apps_backend_tests PRIVATE console_control)
target_compile_options(five_apps_backend_tests PRIVATE -UNDEBUG)
add_executable(console-ctl src/ctl.c)
target_link_libraries(console-ctl PRIVATE console_control)
add_executable(console-standby src/standby.c)
add_executable(console-fileserver src/fileserver.c)
add_executable(console-netplay-peer src/netplay_peer.c)
add_executable(console-link-cable src/link_cable.c ../console-sessiond/src/sha256.c)
target_include_directories(console-link-cable PRIVATE ../console-sessiond/include)
target_compile_definitions(console-link-cable PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
install(TARGETS console-link-cable RUNTIME DESTINATION bin)
target_compile_definitions(console-netplay-peer PRIVATE _GNU_SOURCE)
install(TARGETS console-netplay-peer RUNTIME DESTINATION bin)
install(PROGRAMS init/console-easy-netplay DESTINATION bin)
add_executable(console-time-sync src/time_sync.c)
target_compile_options(console-time-sync PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
install(TARGETS console-time-sync RUNTIME DESTINATION bin)
add_executable(console-timezone src/timezone_main.c)
target_link_libraries(console-timezone PRIVATE console_control)
install(TARGETS console-timezone RUNTIME DESTINATION bin)
target_compile_options(console-fileserver PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
foreach(target console_control console-supervisor console-launcher GameSwitcher console-ctl console-catalog console-standby)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow
            -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef -Werror)
    endif()
endforeach()
install(TARGETS console-supervisor console-launcher GameSwitcher console-sessiond console-catalog console-standby RUNTIME DESTINATION bin)
# Bootstrap helpers must be on the original root image, available before/after card mounts.
install(TARGETS console-ctl RUNTIME DESTINATION bootstrap/usr/magicx/bin)
install(PROGRAMS init/rc.init DESTINATION bootstrap/usr/magicx/bin RENAME console-rc)
install(PROGRAMS init/console-poweroff DESTINATION bootstrap/usr/magicx/bin)
install(PROGRAMS init/shutdown-hook.sh DESTINATION bootstrap/etc/console)
install(FILES config/cards.conf DESTINATION bootstrap/etc/console
        PERMISSIONS OWNER_READ OWNER_WRITE)
install(FILES config/supervisor.json config/base-profile.json config/bridge-launch.json config/ui.json
        DESTINATION bootstrap/etc/console)
if(BUILD_TESTING)
    enable_testing()
    add_executable(system_icon_tests tests/test_system_icons.c)
    target_link_libraries(system_icon_tests PRIVATE console_launcher)
    target_compile_options(system_icon_tests PRIVATE -UNDEBUG)
    add_executable(public_boot_tests tests/test_public_boot.c src/config.c src/catalog_index.c)
    target_link_libraries(public_boot_tests PRIVATE console_control console_launcher)
    target_compile_options(public_boot_tests PRIVATE -UNDEBUG)
    add_executable(splash_fill_tests ../console-packaging/tests/test_splash_fill.c)
    target_link_libraries(splash_fill_tests PRIVATE console_launcher)
    target_compile_options(splash_fill_tests PRIVATE -UNDEBUG)
    add_executable(r18_catalog_tests tests/test_r18_catalog.c src/disk.c src/config.c src/catalog_index.c src/diagnostics.c src/tools_backend.c)
    target_link_libraries(r18_catalog_tests PRIVATE console_control)
    target_compile_options(r18_catalog_tests PRIVATE -UNDEBUG)
    add_executable(probe_files_tests tests/test_probe_files.c src/catalog_index.c
                   ../retroarch-magicx-bridge/magicx_asset_cache.c)
    target_link_libraries(probe_files_tests PRIVATE console_control)

    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(controller_persistence_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME controller_persistence COMMAND controller_persistence_tests
             "${CMAKE_CURRENT_BINARY_DIR}/controller-persistence-fixture")
    # Requires an isolated prepared FAT32/GB/GBC fixture; invoked by the target VM runner.
    add_executable(catalog_runtime_tests tests/test_catalog_runtime.c
                   src/disk.c src/config.c src/catalog_index.c src/diagnostics.c src/tools_backend.c)
    target_link_libraries(catalog_runtime_tests PRIVATE console_control)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(catalog_runtime_tests PRIVATE -UNDEBUG)
    endif()

add_executable(console-boot-status ../console-packaging/image/console-boot-status.c)
target_link_libraries(console-boot-status PRIVATE console_launcher)
