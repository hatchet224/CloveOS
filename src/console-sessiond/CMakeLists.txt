cmake_minimum_required(VERSION 3.16)
project(console_sessiond VERSION 0.2.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SESSIOND_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
option(SESSIOND_SANITIZERS "Enable host address/undefined-behavior sanitizers" OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(SESSIOND_LINUX_DEFAULT ON)
else()
    set(SESSIOND_LINUX_DEFAULT OFF)
endif()
option(SESSIOND_BUILD_DAEMON "Build the real Linux adapters and daemon" ${SESSIOND_LINUX_DEFAULT})

add_library(sessiond_options INTERFACE)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sessiond_options INTERFACE
        -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wstrict-prototypes
        -Wmissing-prototypes -Wformat=2 -Wundef)
    if(SESSIOND_WARNINGS_AS_ERRORS)
        target_compile_options(sessiond_options INTERFACE -Werror)
    endif()
    if(SESSIOND_SANITIZERS)
        if(CMAKE_CROSSCOMPILING)
            message(FATAL_ERROR "Sanitizers are for native host testing only")
        endif()
        target_compile_options(sessiond_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(sessiond_options INTERFACE -fsanitize=address,undefined)
    endif()
endif()

add_library(sessiond_core STATIC src/fsm.c src/power_input.c src/json.c src/sha256.c
    src/document.c src/metadata.c src/hotkeys.c)
target_include_directories(sessiond_core PUBLIC include)
target_link_libraries(sessiond_core PUBLIC sessiond_options)
add_library(sessiond_fake STATIC src/fake_bridge.c src/commands.c)
target_link_libraries(sessiond_fake PUBLIC sessiond_core)

if(SESSIOND_BUILD_DAEMON)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The epoll executable requires Linux; build portable tests on other hosts")
    endif()
    set(THREADS_PREFER_PTHREAD_FLAG ON)
    find_package(Threads REQUIRED)
    add_library(sessiond_linux STATIC src/store.c src/bridge.c src/evdev.c src/profile.c src/worker.c src/library.c src/disc_playlist.c)
    target_compile_definitions(sessiond_linux PUBLIC _GNU_SOURCE _FILE_OFFSET_BITS=64)
    target_link_libraries(sessiond_linux PUBLIC sessiond_core Threads::Threads)
    # Tina glibc 2.29 keeps pthread symbols in a separate library. Keep that
    # dependency explicit even when validating with a newer host compiler.
    target_link_libraries(sessiond_linux PUBLIC pthread)
    add_executable(console-sessiond src/main.c)
    target_link_libraries(console-sessiond PRIVATE sessiond_linux)
    add_executable(console-sessiond-mock src/mock_main.c)
    target_compile_definitions(console-sessiond-mock PRIVATE _GNU_SOURCE)
    target_link_libraries(console-sessiond-mock PRIVATE sessiond_fake)
    install(TARGETS console-sessiond RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(sessiond_tests tests/test_sessiond.c)
    target_link_libraries(sessiond_tests PRIVATE sessiond_fake)
    add_executable(sessiond_standby_tests tests/test_standby.c)
    target_link_libraries(sessiond_standby_tests PRIVATE sessiond_core)
    add_executable(sessiond_library_data_tests tests/test_library_data.c)
    target_link_libraries(sessiond_library_data_tests PRIVATE sessiond_core)
    if(SESSIOND_BUILD_DAEMON)
        add_executable(sessiond_linux_tests tests/test_linux.c)
        target_link_libraries(sessiond_linux_tests PRIVATE sessiond_linux)
        add_executable(sessiond_library_linux_tests tests/test_library_linux.c)
        target_link_libraries(sessiond_library_linux_tests PRIVATE sessiond_linux)
    endif()
    if(NOT CMAKE_CROSSCOMPILING OR CMAKE_CROSSCOMPILING_EMULATOR)
        add_test(NAME sessiond_unit COMMAND sessiond_tests)
        add_test(NAME sessiond_standby COMMAND sessiond_standby_tests)
        add_test(NAME library_data COMMAND sessiond_library_data_tests)
        if(SESSIOND_BUILD_DAEMON)
            add_test(NAME production_adapters COMMAND sessiond_linux_tests)
            add_test(NAME library_linux COMMAND sessiond_library_linux_tests)
            set_tests_properties(production_adapters PROPERTIES TIMEOUT 30)
            foreach(scenario shutdown suspend save_failure)
                set(scenario_options)
                if(scenario STREQUAL "suspend")
                    list(APPEND scenario_options --mock-suspend)
                endif()
                add_test(NAME mock_${scenario} COMMAND console-sessiond-mock
                    ${scenario_options} --script ${CMAKE_CURRENT_SOURCE_DIR}/tests/scenarios/${scenario}.txt)
                set_tests_properties(mock_${scenario} PROPERTIES TIMEOUT 10 RUN_SERIAL TRUE)
            endforeach()
        endif()
    endif()
endif()
